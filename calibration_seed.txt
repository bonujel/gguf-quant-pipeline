def quantize(weights, bits):
    """Simple symmetric quantization of a weight tensor to the given bit width."""
    qmax = (1 << (bits - 1)) - 1
    scale = weights.abs().max() / qmax
    q = (weights / scale).round().clamp(-qmax, qmax)
    return q.to(int), scale

def softmax(x):

class RingBuffer:
    def __init__(self, capacity):
        self.capacity = capacity
        self.data = []
    def push(self, item):
        if len(self.data) >= self.capacity:
            self.data.pop(0)
        self.data.append(item)

func fibonacci(n int) int {
    if n < 2 {
        return n
    }
    a, b := 0, 1
    for i := 2; i <= n; i++ {
        a, b = b, a+b
    }
    return b
}
